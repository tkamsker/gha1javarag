  Usage Example

  from pathlib import Path
  from codeindex.services.ollama_client import create_ollama_client
  from codeindex.services.db_analyzer import DatabaseAnalyzer

  # Create Ollama client
  ollama_client = create_ollama_client()

  # Create analyzer
  analyzer = DatabaseAnalyzer(
      ollama_client=ollama_client,
      output_dir=Path("./output"),
      source_dir=Path("/path/to/java/source"),
      max_workers=10,
      llm_timeout=120,
      max_retries=3,
      force_refresh=False
  )

  # Analyze database layer
  summary = analyzer.analyze_database_layer()

  print(f"Analyzed {summary['analyzed']} files")
  print(f"Extracted {summary['entities']} entities")
  print(f"Extracted {summary['rules']} business rules")
