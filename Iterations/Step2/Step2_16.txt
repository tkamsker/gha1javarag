  Usage Workflow

  # Step 1: Setup environment (one time)

  # Step 2: Index codebase (per project)
  ./run.sh myproject /path/to/source

  # Step 3: Generate PRD documentation
  ./step2.sh myproject /path/to/source ./output/myproject-prd

  # View results
  cat ./output/myproject-prd/master_prd.md
  open ./output/myproject-prd/master_prd.md
